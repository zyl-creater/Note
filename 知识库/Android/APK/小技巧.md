```shell
[zhuyuelin@rd-122:/SDK-SRC/out/target/product/coffey/obj]$ find . -name "*EnergyModeOverlay*"
./METAETC/EnergyModeOverlay_intermediates
./METAETC/EnergyModeOverlay_intermediates/EnergyModeOverlay.meta_lic
[zhuyuelin@rd-122:/SDK-SRC/out/target/product/coffey/obj]$
```
